cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
set(PICO_BOARD pico2_w CACHE STRING "Target board")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(machikania_bt_keyboard C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
foreach(profile classic apple ble)
    set(target machikania_keyboard_${profile})
    set(transport classic)
    if(profile STREQUAL "ble")
        set(transport ble)
    endif()
    add_executable(${target} src/main.c src/keyboard.c src/usb.c src/bluetooth_${transport}.c)
    target_include_directories(${target} PRIVATE src)
    target_link_libraries(${target} pico_stdlib pico_unique_id pico_btstack_${transport}
        pico_btstack_cyw43 pico_cyw43_arch_poll tinyusb_device hardware_watchdog)
    target_compile_definitions(${target} PRIVATE CYW43_LWIP=0)
    if(profile STREQUAL "apple")
        target_compile_definitions(${target} PRIVATE BRIDGE_APPLE_KEYBOARD=1)
    endif()
    if(profile STREQUAL "ble")
        target_sources(${target} PRIVATE src/ble_advertisement.c)
        pico_btstack_make_gatt_header(${target} PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src/bridge.gatt)
    endif()
    pico_enable_stdio_usb(${target} 0)
    pico_enable_stdio_uart(${target} 1)
    pico_add_extra_outputs(${target})
endforeach()
